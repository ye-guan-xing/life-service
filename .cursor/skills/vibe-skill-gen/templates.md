## Template 2: Component `.skill.ts`

Path: `<source-dir>/<ComponentName>.skill.ts`

Generate only when `@skill:runtime` comment or `"runtime": true`.

```typescript
import skillDef from './Login.skill.json'

export type SkillButton = {
  id: string
  selector: string
  cli: string
  handler: string
}

export type SkillRoute = {
  id: string
  from?: string
  to: string
  cli: string
}

export type SkillApi = {
  id: string
  method: string
  url: string
  cli: string
}

export type SkillDefinition = {
  version: number
  component: string
  framework: 'react' | 'vue'
  source: string
  domain: string
  page: string
  buttons: SkillButton[]
  routes: SkillRoute[]
  apis: SkillApi[]
  cli: { name: string; type: string; ref: string }[]
}

export function defineSkill<T extends SkillDefinition>(def: T): T {
  return def
}

export const skill = defineSkill(skillDef as SkillDefinition)

export function getCli(name: string) {
  return skill.cli.find((c) => c.name === name)
}

export function getButton(id: string) {
  return skill.buttons.find((b) => b.id === id)
}

export function getRoute(id: string) {
  return skill.routes.find((r) => r.id === id)
}

export function getApi(id: string) {
  return skill.apis.find((a) => a.id === id)
}

export default skill
```

For React projects using `resolveJsonModule: false`, inline the JSON content instead of import.

## Template 4: Registry README

Path: `.cursor/skills-registry/README.md`

Regenerate entirely on each sync. Do not hand-edit.

```markdown
<!-- AUTO-GENERATED by vibe-skill-gen — do not edit manually -->

# Skills Registry

Updated: 2026-05-19T12:00:00.000Z

## login

Source: `pages/login.vue`

| CLI | Type | Ref | Description |
|-----|------|-----|-------------|
| auth:login:login | button | buttons.login | Click login button |
| auth:login:reset-password | button | buttons.reset | Click reset password button |
| auth:login:go-dashboard | route | routes.login-to-dashboard | Navigate to dashboard |
| auth:login:post-login | api | apis.post-login | POST /api/login |

## checkout

Source: `src/pages/Checkout.tsx`

| CLI | Type | Ref | Description |
|-----|------|-----|-------------|
| cart:checkout:submit | button | buttons.submit | Submit checkout form |
| cart:checkout:go-success | route | routes.checkout-to-success | Navigate to success page |
| cart:checkout:place-order | api | apis.place-order | POST /api/orders |
```

### README Generation Rules

1. One `## <slug>` section per registry JSON file
2. List all `components[].source` under section header
3. Flatten all `cli` entries from all components in that slug
4. Sort CLI rows alphabetically by `name`
5. Update top `Updated:` timestamp to latest registry `updatedAt`
